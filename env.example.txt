# BudgetRail — environment variable documentation
# Copy to .env and fill in values. NEVER commit .env.
# Public chain and contract config is safe to share; private keys NEVER belong here.
#
# Network selection (build-time):
#   VITE_ARC_NETWORK=testnet  → Arc Testnet (chain 5042002)  [default]
#   VITE_ARC_NETWORK=mainnet  → Arc Mainnet (chain 5042)
# The frontend refuses to make writes if the wallet is on the wrong chain.

# ─── Network selection ────────────────────────────────────────────────────────
VITE_ARC_NETWORK=testnet

# ─── Testnet frontend (VITE_ARC_NETWORK=testnet) ─────────────────────────────
# Vite exposes VITE_* vars to the browser bundle — public values only.

# BudgetRail contract address on Arc Testnet
VITE_BUDGET_RAIL_ADDRESS=0xf6a13df15ad93c2e2968e1d69c53894a923bda21

# Provider API base URL — NO trailing slash, NO /api suffix
# Use "" (empty) when the Vite dev proxy handles /api/* → localhost:3001
# Use "https://your-provider.example.com" for production
VITE_PROVIDER_API_URL=

# Provider payout wallet address (role C — receives USDC)
VITE_PROVIDER_PAYOUT=0x0000000000000000000000000000000000000000

# Quote signer PUBLIC address (role D public key — safe to bundle)
VITE_QUOTE_SIGNER=0x0000000000000000000000000000000000000000

# ─── Mainnet frontend (VITE_ARC_NETWORK=mainnet) ─────────────────────────────
# STOP: Do not set mainnet vars without completing the mainnet runbook in
# docs/GRANT-EVIDENCE.md and obtaining an independent security review first.

# BudgetRail contract address on Arc Mainnet (set after mainnet deployment)

# VITE_MAINNET_PROVIDER_PAYOUT=0x...

# Mainnet quote signer PUBLIC address (role D)
# VITE_MAINNET_QUOTE_SIGNER=0x...
